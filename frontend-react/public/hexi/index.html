3. Chatbot Integration Code

How to use Hexi in your Chatbot project:

Include hexi-bot.js or embed the standalone SVG files directly into your HTML/React/Vue app.

<!-- 1. Include Script -->
<script src="hexi-bot.js"></script>

<!-- 2. Embed Component -->
<hexi-bot id="myAvatar" state="idle" theme="dark"></hexi-bot>

<!-- 3. Control in JavaScript -->
<script>
  const bot = document.getElementById('myAvatar');

  // Trigger base states
  bot.setState('idle');   // Idle base state
  bot.setState('happy');  // Happy base state

  // Trigger searching with customizable loops
  bot.runSearchSequence({
    lookupLoops: 3,  // Number of 1.2s scan loops while fetching API
    textLoops: 2,    // Number of text animation loops
    onComplete: () => {
      bot.setState('happy'); // Transition to happy when answer finishes!
    }
  });
</script>
